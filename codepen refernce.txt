gsap.registerPlugin(SplitText, ScrollTrigger);

$(function () {
  const $titles = $(".title-animation");

  $titles.each(function () {
    const $title = $(this);

    const split = new SplitText($title[0], { type: "lines" });
    const lines = split.lines;
    
    $title.empty();

    lines.forEach(line => {
      const mask = document.createElement("div");
      mask.className = "mask";

      const inner = document.createElement("div");
      inner.className = "line-inner";
      inner.textContent = line.textContent;

      mask.appendChild(inner);
      $title[0].appendChild(mask);
    });

    gsap.from($title.find(".mask .line-inner"), {
      y: 50,
      opacity: 0,
      duration: 0.7,
      ease: "power3.out",
      stagger: 0.15,
      scrollTrigger: {
        trigger: $title[0],
        start: "top 70%",
        toggleActions: "play none play reverse"
      }
    });
  });
});
